 SELECT 'SERVICES' AS MAJDEP,
 CASE WHEN DEPID IN (SELECT DEPT_DEPID FROM HS_DEPT_DEPARTMENT WHERE DEPT_DEPTYPE='HISTOLOGY') THEN 'PATHOLOGY LAB' ELSE 'OTHERS' END AS MAINDEP,
 DEPNAME,SUM(QTY) QTY,SUM(TOTALAMOUNT)TOTALAMOUNT, SUM(TOTALDISCOUNT) TOTALDISCOUNT,
 SUM(GROSSAMOUNT)GROSSAMOUNT,SUM(HST) HST,SUM(GROSSAMOUNT+HST) NETTOTAL
 FROM VW_NewDept_Wise_Total_Coll
 where BillDate Between '2076/05/12'and '2076/05/12'
 Group by depid,DepName
 Order by DepName
